pipeline {
    agent any

    options {
        timestamps()
        buildDiscarder(logRotator(numToKeepStr: '10', artifactNumToKeepStr: '10'))
    }

    stages {
        stage('Checkout') {
            steps {
                git branch: 'main', url: 'https://github.com/kailash1306/agile-assessment.git'
            }
        }

        stage('Parallel Checks') {
            steps {
                script {
                    def startedAt = System.currentTimeMillis()

                    parallel(
                        frontend: {
                            sh 'python frontend.py'
                        },
                        backend: {
                            sh 'python backend.py'
                        }
                    )

                    def elapsedSeconds = (System.currentTimeMillis() - startedAt) / 1000.0
                    echo "Parallel checks completed in approximately ${elapsedSeconds} seconds."
                    echo 'Expected parallel time: about 4 seconds plus Jenkins overhead.'
                    echo 'Expected serial time: about 8 seconds plus Jenkins overhead (4 seconds per script).'
                }
            }
        }

        stage('Archive Reports') {
            steps {
                archiveArtifacts artifacts: 'frontend_report.txt,backend_report.txt', fingerprint: true
                echo "Archived frontend_report.txt and backend_report.txt for build ${env.BUILD_NUMBER}."
                echo 'To verify retention: run a newer build, then open the older build and its Artifacts section.'
                echo 'Because artifactNumToKeepStr is 10, older build reports remain available until that retention limit is reached .'
            }
        }
    }
}
